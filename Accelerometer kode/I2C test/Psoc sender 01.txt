Virker til at se om psoc vil sende

#include "project.h"

int main(void)
{
    CyGlobalIntEnable;

    I2C_Start();

    for(;;)
    {
        uint8 status;

        status = I2C_MasterSendStart(0x4A, I2C_WRITE_XFER_MODE);

        if(status == I2C_MSTR_NO_ERROR)
        {
            I2C_MasterWriteByte(0x01);
        }

        I2C_MasterSendStop();

        CyDelay(100);
    }
}
